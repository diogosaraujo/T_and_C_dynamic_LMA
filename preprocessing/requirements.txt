cdsapi>=0.7.4

# Verification step (verify_era5_land.py) -- reads the downloaded netCDFs.
xarray>=2024.1.0

# AmeriFlux BADM inspection -- BADM/BIF metadata ships as .xlsx.

openpyxl>=3.1

# Gap-filling from gridded products (fetch_canopy_height.py and the soil/root
# fetchers to come). rasterio bundles GDAL, which reads the remote mosaics over
# /vsicurl/ so multi-GB rasters never have to be downloaded.
rasterio>=1.3

# Time-varying LMA (build_lma_input.py). The PLSR pipeline saves its outputs with
# MATLAB's '-v7.3', which is HDF5 and unreadable by scipy.io.loadmat -- hence h5py.
# scipy stays in the list because an older run could have produced a v7 file, and
# the loader falls back to it automatically.
h5py>=3.8
scipy>=1.10

# Ecoregion pairing check (verify_ecoregion_pairing.py). pyshp is pure Python and
# reads the EPA Level III shapefile without GDAL; geopandas would drag in fiona,
# shapely and pyproj for one point-in-polygon test. The Albers projection is
# implemented from the shapefile's own .prj and validated against control stations.
pyshp>=2.3

# Analysis and figures. pandas does the group-bys in summarise_effect.py over
# tables of several million rows; matplotlib renders the skill maps and Taylor
# diagrams. Their absence is what killed jobs 39623-39625, which had already
# read every model table, every FLUXNET archive and the SPEI stacks before
# dying on "import matplotlib" at the plotting call.
#
# pyproj gives the figures CONUS Albers without pulling in cartopy; geopandas is
# optional and only used for the --basemap outline, so it is deliberately NOT
# listed -- the maps draw without it.
pandas>=2.0
matplotlib>=3.7
pyproj>=3.5
